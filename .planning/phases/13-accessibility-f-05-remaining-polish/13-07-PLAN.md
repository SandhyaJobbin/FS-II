        custom={6}

              t.verdict === 'correct' ? 'bg-emerald-50 border-emerald-200' :
              t.verdict === 'incorrect' ? 'bg-red-50 border-red-200' : 'bg-amber-50 border-amber-200';

                    <span className="font-medium">{t.questionStem || t.qId}</span>

                      {t.verdict}
                    </span>
                  </summary>
                  <ul className="mt-2 ml-4 flex flex-col gap-1 text-xs text-slate-700">
                    {t.criteriaMet.map((c) => (

                        <span aria-hidden="true" className={c.met ? 'text-emerald-600' : 'text-red-600'}>
